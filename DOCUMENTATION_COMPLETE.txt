4.1 CONFIGURATION DES PINS (Setup)

void setup() {
  // Entrées avec pull-up interne
  pinMode(BTN_START_PIN, INPUT_PULLUP);      // GPIO 0
  pinMode(BTN_STOP_PIN, INPUT_PULLUP);       // GPIO 1
  pinMode(BTN_EMERGENCY_PIN, INPUT_PULLUP);  // GPIO 2
  pinMode(SW_AUTO_MANUAL_PIN, INPUT_PULLUP); // GPIO 3
  pinMode(SW_DIRECTION_PIN, INPUT_PULLUP);   // GPIO 4
  pinMode(ENCODER_CLK_PIN, INPUT_PULLUP);    // GPIO 5
  pinMode(ENCODER_DT_PIN, INPUT_PULLUP);     // GPIO 6
  pinMode(ENCODER_SW_PIN, INPUT_PULLUP);     // GPIO 7
  pinMode(PAL_INPUT_PIN, INPUT);             // GPIO 26 (ADC)

  // Sorties moteur
  pinMode(STEPPER_STEP_PIN, OUTPUT);         // GPIO 8
  pinMode(STEPPER_DIR_PIN, OUTPUT);          // GPIO 9
  pinMode(STEPPER_ENABLE_PIN, OUTPUT);       // GPIO 10

  // I2C Écran OLED (automatique via Wire.begin())
  // GPIO 12 (SDA) + GPIO 13 (SCL)

  // Initialiser écran OLED à 0x3C
  if (!display.begin(SSD1306_SWITCHCAPVCC, OLED_ADDR)) {
    Serial.println("SSD1306 allocation failed");
    while(1);
  }
}

4.2 BOUCLE PRINCIPALE (Loop)
─────────────────────────────────────────────────────────────────────────────

void loop() {
  // 1. Lecture des entrées (tous les 10ms)
  readInputs();
  
  // 2. Gestion des événements
  handleButtons();
  handleManualEncoder();
  
  // 3. Calcul de la fréquence cible
  updateTargetFrequency();
  
  // 4. Rampe d'accélération/décélération
  handleAcceleration();
  
  // 5. Génération des pulses STEP
  generateSteps();
  
  // 6. Mise à jour écran OLED (toutes les 200ms)
  updateDisplay();
  
  // Délai de polling
  delay(10);
}

4.3 GESTION DES RAMPES (Accélération/Décélération)
─────────────────────────────────────────────────────────────────────────────

void handleAcceleration() {
  unsigned long now = millis();
  
  if (isAccelerating) {
    unsigned long elapsed = now - accelStartTime;
    
    if (elapsed >= ACCEL_TIME) {
      // Accélération terminée
      isAccelerating = false;
      currentFreq = targetFreq;
    } else {
      // Rampe linéaire d'accélération
      float progress = (float)elapsed / ACCEL_TIME;
      currentFreq = targetFreq * progress;
    }
  }
  else if (isDecelerating) {
    unsigned long decelDuration = isEmergencyStopping ? 
                                   EMERGENCY_DECEL_TIME : 
                                   DECEL_TIME;
    unsigned long elapsed = now - decelStartTime;
    
    if (elapsed >= decelDuration) {
      // Décélération terminée
      isDecelerating = false;
      currentFreq = 0;
      digitalWrite(STEPPER_ENABLE_PIN, HIGH);  // Désactiver driver
    } else {
      // Rampe linéaire de décélération
      float progress = 1.0 - ((float)elapsed / decelDuration);
      currentFreq = targetFreq * progress;
    }
  }
}

4.4 GÉNÉRATION DES SIGNAUX MOTEUR (STEP)
─────────────────────────────────────────────────────────────────────────────

void generateSteps() {
  if (currentFreq <= 0) {
    stepInterval = 0;
    return;
  }
  
  // Calcul de l'intervalle entre les steps (µs)
  // Fréquence = 2 steps par cycle (HIGH et LOW)
  stepInterval = (unsigned long)(1000000.0 / (currentFreq * 2));
  
  // Génération des steps via polling
  if (motorEnabled) {
    unsigned long now = micros();
    if ((now - lastStepTime) >= stepInterval) {
      // Pulse STEP
      digitalWrite(STEPPER_STEP_PIN, HIGH);
      delayMicroseconds(5);
      digitalWrite(STEPPER_STEP_PIN, LOW);
      lastStepTime = now;
    }
  }
}

4.5 SYNCHRONISATION PAL (Mode AUTO)
─────────────────────────────────────────────────────────────────────────────

// ISR pour détection PAL
void palDetectISR() {
  unsigned long now = micros();
  
  if (lastPalTime > 0) {
    unsigned long interval = now - lastPalTime;
    // Filtre: PAL = 20ms ± 10%
    if (interval > 18000 && interval < 22000) {
      palFreq = 1000000.0 / interval;
    }
  }
  lastPalTime = now;
}

// Maintien de l'alignement de phase
void maintainPhaseAlignment() {
  float syncTolerance = targetFreq * 0.02;  // ±2%
  
  if (currentFreq < (targetFreq - syncTolerance)) {
    currentFreq += 10;  // Ajustement ascendant
  } else if (currentFreq > (targetFreq + syncTolerance)) {
    currentFreq -= 10;  // Ajustement descendant
  }
}

4.6 AFFICHAGE OLED (200ms)
─────────────────────────────────────────────────────────────────────────────

void updateDisplay() {
  static unsigned long lastDisplayUpdate = 0;
  unsigned long now = millis();
  
  if ((now - lastDisplayUpdate) < 200) return;
  lastDisplayUpdate = now;
  
  display.clearDisplay();
  
  // Zone haute: Fréquence (inverse vidéo)
  display.fillRect(0, 0, 128, 20, SSD1306_WHITE);
  display.setTextColor(SSD1306_BLACK);
  display.setTextSize(2);
  
  char freqStr[20];
  dtostrf(currentFreq, 5, 0, freqStr);
  strcat(freqStr, " Hz");
  
  // Centrer le texte
  int16_t x1, y1;
  uint16_t w, h;
  display.getTextBounds(freqStr, 0, 0, &x1, &y1, &w, &h);
  display.setCursor((128 - w) / 2, 2);
  display.println(freqStr);
  
  // Zone milieu: Mode
  display.setTextColor(SSD1306_WHITE);
  display.setTextSize(2);
  display.setCursor(0, 32);
  const char* modeStr = autoMode ? "AUTO" : "MANUEL";
  display.getTextBounds(modeStr, 0, 0, &x1, &y1, &w, &h);
  display.setCursor((128 - w) / 2, 32);
  display.println(modeStr);
  
  // Zone basse: Direction et statut
  display.setTextSize(1);
  const char* dirStr = directionForward ? "AVANT" : "ARRIERE";
  display.getTextBounds(dirStr, 0, 0, &x1, &y1, &w, &h);
  display.setCursor((128 - w) / 2, 56);
  display.println(dirStr);
  
  // Statut moteur
  display.setCursor(0, 56);
  display.println(motorEnabled ? "[ON]" : "[OFF]");
  
  display.display();
}

SECTION 8: NOTES ET AVERTISSEMENTS

⚠️  AVERTISSEMENTS IMPORTANTS
───────────────────────────────────────────────────────────────────────────────

1. SÉCURITÉ ÉLECTRIQUE:
   - Vérifier toutes les polarités avant mise sous tension
   - Moteur en marche: risque de pincement
   - Courant moteur: peut atteindre 3A, assurez une bonne ventilation
   - Ne pas toucher le moteur pendant le fonctionnement

2. SYNCHRONISATION PAL:
   - Nécessite un signal 50Hz stable (ex: caméra PAL)
   - La qualité du signal affecte la précision
   - Filtre RC requis pour l'entrée ADC

3. ENCODEUR ROTATIF:
   - Mode fine: très sensible (1 Hz par cran)
   - Risque de dépassement à haute fréquence
   - Debounce logiciel: 200ms

4. MOTEUR NEMA17:
   - Couple limité (0.4 Nm nominale)
   - Peut chauffer à fréquence élevée
   - Vérifier l'isolation thermique du driver

5. COMPILATEUR EARLEPHILHOWER:
   - Compatible Arduino IDE 2.0+
   - Non testé avec Arduino IDE 1.8 (ancienne version)
   - Mettre à jour régulièrement pour les correctifs

LIMITATIONS CONNUES
───────────────────────────────────────────────────────────────────────────────

1. Fréquence maximale: 20000 Hz (limitation du CPU @ 125MHz)
2. Résolution de fréquence: ~1 Hz minimum
3. Précision PAL: ±2% (dépendant du signal d'entrée)
4. Écran OLED: Pas de support couleur
5. Pas de retour d'encodeur (pas de confirmation mécanique)
